cmake_minimum_required(VERSION 3.20)
project(StableVQADeployment LANGUAGES CXX)
if(POLICY CMP0144)
  cmake_policy(SET CMP0144 NEW)
endif()

set(STABLEVQA_PLATFORM "AUTO" CACHE STRING "AUTO, X86, MACOS, or GPU")
set_property(CACHE STABLEVQA_PLATFORM PROPERTY STRINGS AUTO X86 MACOS GPU)
option(STABLEVQA_GPU "Build the Linux CUDA/TensorRT implementation" OFF)
set(ONNXRUNTIME_ROOT "" CACHE PATH "ONNX Runtime installation root")

if(APPLE AND NOT ONNXRUNTIME_ROOT AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/onnxruntime-macos/include/onnxruntime_cxx_api.h")
  set(ONNXRUNTIME_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/onnxruntime-macos")
endif()

if(STABLEVQA_PLATFORM STREQUAL "AUTO")
  if(STABLEVQA_GPU)
    set(STABLEVQA_PLATFORM "GPU")
  elseif(APPLE)
    set(STABLEVQA_PLATFORM "MACOS")
  else()
    set(STABLEVQA_PLATFORM "X86")
  endif()
endif()

if(STABLEVQA_PLATFORM STREQUAL "GPU" AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The GPU implementation is supported only on Linux.")
endif()
if(STABLEVQA_PLATFORM STREQUAL "MACOS" AND NOT APPLE)
  message(FATAL_ERROR "MACOS implementation selected on a non-macOS host.")
endif()
if(NOT STABLEVQA_PLATFORM MATCHES "^(X86|MACOS|GPU)$")
  message(FATAL_ERROR "STABLEVQA_PLATFORM must be AUTO, X86, MACOS, or GPU.")
endif()

list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
find_package(OpenCV REQUIRED)
find_package(ONNXRuntime REQUIRED)

set(STABLEVQA_SOURCE "")
if(STABLEVQA_PLATFORM STREQUAL "GPU")
  set(STABLEVQA_SOURCE platform/gpu/StableVqaGpu.cpp)
elseif(STABLEVQA_PLATFORM STREQUAL "MACOS")
  set(STABLEVQA_SOURCE platform/macos/StableVqaMac.cpp)
else()
  set(STABLEVQA_SOURCE platform/windows/StableVqaX86.cpp)
endif()

add_executable(stablevqa ${STABLEVQA_SOURCE})
target_compile_features(stablevqa PRIVATE cxx_std_17)
target_include_directories(stablevqa PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(stablevqa PRIVATE ONNXRuntime::ONNXRuntime ${OpenCV_LIBS})

if(MSVC)
  target_compile_options(stablevqa PRIVATE /O2 /W4 /favor:INTEL64)
else()
  target_compile_options(stablevqa PRIVATE -O3 -march=native -ffast-math -Wall -Wextra)
  if(STABLEVQA_PLATFORM STREQUAL "MACOS")
    target_compile_options(stablevqa PRIVATE -flto)
    target_link_options(stablevqa PRIVATE -flto)
  endif()
endif()

if(STABLEVQA_PLATFORM STREQUAL "GPU")
  target_compile_definitions(stablevqa PRIVATE STABLEVQA_CUDA=1)
endif()

if(APPLE)
  set(_stablevqa_rpath "@executable_path/../onnxruntime/lib")
elseif(UNIX)
  set(_stablevqa_rpath "$ORIGIN/../onnxruntime/lib")
endif()
if(DEFINED _stablevqa_rpath)
  set_target_properties(stablevqa PROPERTIES
    BUILD_RPATH "${ONNXRUNTIME_ROOT}/lib"
    INSTALL_RPATH "${_stablevqa_rpath}")
endif()

add_executable(test_cpu_tuning tests/test_cpu_tuning.cpp)
target_compile_features(test_cpu_tuning PRIVATE cxx_std_17)
target_include_directories(test_cpu_tuning PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")

message(STATUS "StableVQA platform implementation: ${STABLEVQA_PLATFORM}")
